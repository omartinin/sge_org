## 1. Personalización del módulo

- Creamos el controller:

    `controllers.py`
    ```python
    # -*- coding: utf-8 -*-
    from odoo import http
    from odoo.http import request, Response
    import json


    class Subscription(http.Controller):
        @http.route('/api/subscription', type='http', methods=['GET'], csrf=False)
        def get_subscriptions(self, **kwargs):
            try:
                ALLOWED_STATUS = ['active', 'expired', 'pending', 'cancelled']
                status = request.params.get('status')

                if status and status not in ALLOWED_STATUS:
                    return Response(
                        json.dumps({
                            "error": f"Estado no permitido: {status}",
                            "allowed_values": ALLOWED_STATUS
                        }),
                        content_type="application/json",
                        status=400
                    )
                if status:
                    subscripciones = request.env['subscription.subscription'].search([('status','=',status)])
                else:
                    subscripciones = request.env['subscription.subscription'].search([])
                data = []
                for sub in subscripciones:
                    data.append({
                        "id": sub.id,
                        "name": sub.name,
                        "customer_id": sub.customer_id.id,
                        "customer_name": sub.customer_id.name,
                        "subscription_code": sub.subscription_code,
                        "start_date": sub.start_date.isoformat() if sub.start_date else None,
                        "end_date": sub.end_date.isoformat() if sub.end_date else None,
                        "duration_months": sub.duration_months,
                        "renewal_date": sub.renewal_date.isoformat() if sub.renewal_date else None,
                        "status": sub.status,
                        "is_renewable": sub.is_renewable,
                        "auto_renewal": sub.auto_renewal,
                        "price": sub.price,
                        "usage_limit": sub.usage_limit,
                        "current_usage": sub.current_usage,
                        "use_percent": sub.use_percent
                    })

                return Response(
                    json.dumps(data),
                    content_type="application/json",
                    status=200
                )
            except Exception as e:
                return Response(
                    json.dumps({"error": f"Error interno del servidor: {str(e)}"}),
                    content_type="application/json",
                    status=500
                )

        @http.route('/api/subscription/<string:name>', type='http', methods=['GET'], csrf=False)
        def get_subscription(self, name):
            try:
                subscripciones = request.env['subscription.subscription'].search([('name','=',name)], limit=1)
                
                data = []
                for sub in subscripciones:
                    data.append({
                        "id": sub.id,
                        "name": sub.name,
                        "customer_id": sub.customer_id.id,
                        "customer_name": sub.customer_id.name,
                        "subscription_code": sub.subscription_code,
                        "start_date": sub.start_date.isoformat() if sub.start_date else None,
                        "end_date": sub.end_date.isoformat() if sub.end_date else None,
                        "duration_months": sub.duration_months,
                        "renewal_date": sub.renewal_date.isoformat() if sub.renewal_date else None,
                        "status": sub.status,
                        "is_renewable": sub.is_renewable,
                        "auto_renewal": sub.auto_renewal,
                        "price": sub.price,
                        "usage_limit": sub.usage_limit,
                        "current_usage": sub.current_usage,
                        "use_percent": sub.use_percent
                    })
                return Response(
                    json.dumps(data),
                    content_type="application/json",
                    status=200
                )
            except Exception as e:
                return Response(
                    json.dumps({"error": f"Error interno del servidor: {str(e)}"}),
                    content_type="application/json",
                    status=500
                )
    ```
